# solid-kit, этап 1: Tabs — план реализации

> **Для исполнителей:** ОБЯЗАТЕЛЬНЫЙ суб-скил: superpowers:executing-plans (или subagent-driven-development). Шаги отмечены чекбоксами (`- [ ]`).

**Цель:** одиннадцатый и последний компонент этапа 1. Tabs: compound-компонент `Tabs.Root` / `Tabs.List` / `Tabs.Tab` / `Tabs.Panel` по WAI-ARIA Tabs с автоматической активацией, клавиатурой по кругу, горизонтальной прокруткой полосы, внутренним кольцом фокуса и индикатором выбранной вкладки. Полоса вкладок стоит в одной линии с остальными контролами во всех пяти рядах; контраст текста и индикатора, связи ARIA, прокрутка и кольцо фокуса проверяются на странице; компонент входит в матрицу формы пакета (включая порядок CSS с основой контролов).

**Архитектура:** `Tabs.Root` держит состояние (`createControllableSignal<string | null>`), реестр вкладок (для входа в порядок Tab первой доступной вкладки, когда выбранной нет) и отдаёт всё через контекст. `Tabs.List` (скроллер, `data-s-kit-focus-inset`) обрабатывает клавиши через чистую функцию `nextTabIndex`. `Tabs.Tab` это `button role="tab"` с основой контролов через `composes`, `Tabs.Panel` скрывает невыбранную панель (`hidden`) и не монтирует её содержимое без `keepMounted`. Вкладка прокручивается в видимую область функцией `revealInScroller`, которая двигает только полосу. Идентификаторы строятся функциями `tabId` / `panelId`.

**Стек:** как в этапе 0 и предыдущих планах этапа 1.

**Спеки:** `docs/superpowers/specs/2026-10-29-stage-1-simple-controls-design.md` (основной; раздел 3.10 описывает Tabs, разделы 2 и 4 общие соглашения и страницу, раздел 7 вопрос 3), `docs/superpowers/specs/2026-10-28-solid-kit-umbrella-design.md` (разделы 4.1, 5, 5.1 про внутреннее кольцо и обрезающие контейнеры, 7, 11.1, 12).

/**
 * Индекс вкладки, на которую ведёт клавиша, среди `count` доступных вкладок; `null`, если клавиша не навигационная.
 * Стрелки идут по кругу; если текущая вкладка неизвестна (`current` < 0), ArrowRight начинает с первой, ArrowLeft с последней.
 */
export function nextTabIndex(key: string, current: number, count: number): number | null {
	if (count === 0) {
		return null;
	}
	if (key === 'ArrowRight') {
		return current < 0 ? 0 : (current + 1) % count;
	}
	if (key === 'ArrowLeft') {
		return current < 0 ? count - 1 : (current - 1 + count) % count;
	}
	if (key === 'Home') {
		return 0;
	}
	if (key === 'End') {
		return count - 1;
	}
	return null;
}
```

/** Прокручивает горизонтальный скроллер ровно настолько, чтобы элемент оказался в его видимой области; страницу не двигает. */
export function revealInScroller(item: Element, scroller: HTMLElement): void {
	const itemRect = item.getBoundingClientRect();
	const scrollerRect = scroller.getBoundingClientRect();
	if (itemRect.left < scrollerRect.left) {
		scroller.scrollLeft -= scrollerRect.left - itemRect.left;
	}
	else if (itemRect.right > scrollerRect.right) {
		scroller.scrollLeft += itemRect.right - scrollerRect.right;
	}
}
```

export interface TabsContextValue {
	/** Основа идентификаторов вкладок и панелей этого `Tabs.Root`. */
	baseId: string;
	value: () => string | null;

		throw new Error(`${part} must be used inside Tabs.Root`);

```tsx
import { createMemo, createSignal, createUniqueId, type JSX, splitProps } from 'solid-js';

import { createControllableSignal } from '../../utils/create-controllable-signal';

import { TabsContext, type TabsContextValue } from './tabs-context';
import type { TabEntry } from './tabs-entry';

export interface TabsRootProps extends Omit<JSX.HTMLAttributes<HTMLDivElement>, 'class' | 'style'> {
	value?: string | null | undefined;
	defaultValue?: string | null | undefined;
	onValueChange?: ((value: string) => void) | undefined;
	class?: string | undefined;
	style?: JSX.CSSProperties | undefined;
}

export function TabsRoot(props: TabsRootProps): JSX.Element {
	const [local, rest] = splitProps(props, ['value', 'defaultValue', 'onValueChange', 'class', 'style']);
	const baseId = createUniqueId();

	const [value, setValue] = createControllableSignal<string | null>({
		name: 'Tabs',
		value: () => local.value,
		defaultValue: () => local.defaultValue ?? null,
		onChange: (next) => {
			if (next !== null) {
				local.onValueChange?.(next);
			}
		}
	});

	const [entries, setEntries] = createSignal<readonly TabEntry[]>([]);
	const tabbableValue = createMemo(() => {
		const enabled = entries().filter((entry) => !entry.disabled());
		const selected = enabled.find((entry) => entry.value() === value());
		return (selected ?? enabled[0])?.value() ?? null;
	});

	const context: TabsContextValue = {
		baseId,
		value,
		select: (next) => {
			if (next !== value()) {
				setValue(next);
			}
		},
		tabbableValue,
		register: (entry) => {
			setEntries((list) => [...list, entry]);
			return () => {
				setEntries((list) => list.filter((item) => item !== entry));
			};
		}
	};

	return (
		<TabsContext.Provider value={context}>
			<div {...rest} class={local.class} style={local.style}/>
		</TabsContext.Provider>
	);
}

import styles from './tabs.module.css';

import { type JSX, splitProps } from 'solid-js';

import { useTabsContext } from './tabs-context';
import { nextTabIndex } from './tabs-navigation';

export type TabsListSize = 'sm' | 'md' | 'lg';

export interface TabsListProps extends Omit<JSX.HTMLAttributes<HTMLDivElement>, 'class' | 'style' | 'role' | 'onKeyDown'> {
	size?: TabsListSize | undefined;
	class?: string | undefined;
	style?: JSX.CSSProperties | undefined;
	onKeyDown?: JSX.EventHandler<HTMLDivElement, KeyboardEvent> | undefined;
}

export function TabsList(props: TabsListProps): JSX.Element {
	const tabs = useTabsContext('Tabs.List');
	const [local, rest] = splitProps(props, ['size', 'class', 'style', 'onKeyDown']);

	const handleKeyDown: JSX.EventHandler<HTMLDivElement, KeyboardEvent> = (event) => {
		local.onKeyDown?.(event);
		if (event.defaultPrevented || event.altKey || event.ctrlKey || event.metaKey) {
			return;
		}
		const current = event.target.closest<HTMLElement>('[role="tab"]');
		if (!current) {
			return;
		}
		const enabled = [...event.currentTarget.querySelectorAll<HTMLElement>('[role="tab"]:not(:disabled)')];
		const index = nextTabIndex(event.key, enabled.indexOf(current), enabled.length);
		const target = index === null ? undefined : enabled[index];
		if (!target) {
			return;
		}
		event.preventDefault();
		target.focus();
		const value = target.dataset['tabValue'];
		if (value !== undefined) {
			tabs.select(value);
		}
	};

	return (
		<div
			{...rest}
			role='tablist'
			data-s-kit
			data-s-kit-size={local.size}
			data-s-kit-focus-inset
			class={local.class ? `${styles['list']} ${local.class}` : styles['list']}
			style={local.style}
			onKeyDown={handleKeyDown}
		/>
	);
}

import styles from './tabs.module.css';

import { createEffect, type JSX, on, onCleanup, onMount, splitProps } from 'solid-js';

import { revealInScroller } from './reveal-in-scroller';
import { useTabsContext } from './tabs-context';
import { panelId, tabId } from './tabs-ids';

export interface TabsTabProps extends Omit<JSX.ButtonHTMLAttributes<HTMLButtonElement>, 'class' | 'role' | 'type' | 'onClick' | 'onFocus'> {
	value: string;
	class?: string | undefined;
	onClick?: JSX.EventHandler<HTMLButtonElement, MouseEvent> | undefined;
	onFocus?: JSX.EventHandler<HTMLButtonElement, FocusEvent> | undefined;
}

export function TabsTab(props: TabsTabProps): JSX.Element {
	const tabs = useTabsContext('Tabs.Tab');
	const [local, rest] = splitProps(props, ['value', 'disabled', 'class', 'onClick', 'onFocus']);
	const selected = (): boolean => tabs.value() === local.value;
	let element: HTMLButtonElement | undefined;

	const reveal = (): void => {
		const scroller = element?.closest<HTMLElement>('[role="tablist"]');
		if (element && scroller) {
			revealInScroller(element, scroller);
		}
	};

	onCleanup(tabs.register({ value: () => local.value, disabled: () => local.disabled === true }));
	// выбор сменили не фокусом (например, родитель): вкладка прокручивается в видимую область; выбранная при первом показе тоже
	createEffect(on(selected, (isSelected) => {
		if (isSelected) {
			reveal();
		}
	}, { defer: true }));
	onMount(() => {
		if (selected()) {
			reveal();
		}
	});

	return (
		<button
			{...rest}
			ref={element}
			type='button'
			role='tab'
			id={tabId(tabs.baseId, local.value)}
			aria-selected={selected()}
			aria-controls={panelId(tabs.baseId, local.value)}
			tabindex={tabs.tabbableValue() === local.value ? 0 : -1}
			data-tab-value={local.value}
			disabled={local.disabled}
			class={local.class ? `${styles['tab']} ${local.class}` : styles['tab']}
			onClick={(event) => {
				local.onClick?.(event);
				if (!event.defaultPrevented) {
					tabs.select(local.value);
				}
			}}
			onFocus={(event) => {
				local.onFocus?.(event);
				reveal();
			}}
		/>
	);
}

import { contrastRatio, type Rgb } from '../../src/utils/color-math';

import { type CheckResult, resolveColor, type Rgba } from './theme-checks';

function rgb([red, green, blue]: Rgba): Rgb {
	return [red, green, blue];
}

/** Первый цвет в вычисленном `box-shadow` (браузер пишет его в любой из своих нотаций). */
const COLOR_IN_SHADOW = /(?:rgba?|color|oklch|oklab|lab|lch|hsla?)\([^)]*\)/;

/**
 * У каждой полосы с data-tabs-demo: роль, одна выбранная вкладка и одна видимая панель, связи вкладок и панелей по id,
 * контраст текста выбранной и невыбранной вкладок (4.5:1) и индикатора (3:1) к поверхности. У полосы `scroll` дополнительно:
 * содержимое шире полосы, прокрутка включена и вкладки занимают всю высоту полосы (полоса прокрутки место не занимает).
 */
export function runTabsChecks(page: HTMLElement): CheckResult[] {
	const results: CheckResult[] = [];
	const check = (name: string, ok: boolean, detail: string): void => {
		results.push({ name, ok, detail });
	};
	const surface = resolveColor(getComputedStyle(page).backgroundColor);
	const ratio = (color: string): number => contrastRatio(rgb(resolveColor(color)), rgb(surface));

	for (const list of page.querySelectorAll<HTMLElement>('[data-tabs-demo]')) {
		const id = list.dataset['tabsDemo'] ?? '?';
		const tabs = [...list.querySelectorAll<HTMLElement>('[role="tab"]')];
		const selected = tabs.filter((item) => item.getAttribute('aria-selected') === 'true');
		const plain = tabs.find((item) => item.getAttribute('aria-selected') === 'false' && !item.hasAttribute('disabled'));
		const firstSelected = selected[0];
		const root = list.parentElement;
		if (!firstSelected || !plain || !root) {
			throw new Error(`Playground element is missing in Tabs ${id}`);
		}

		check(`Tabs ${id}: роль tablist`, list.getAttribute('role') === 'tablist', `role = ${list.getAttribute('role') ?? 'null'}`);
		check(`Tabs ${id}: выбрана одна вкладка`, selected.length === 1, `выбрано ${selected.length} из ${tabs.length}`);
		const visiblePanels = root.querySelectorAll('[role="tabpanel"]:not([hidden])').length;
		check(`Tabs ${id}: видна одна панель`, visiblePanels === 1, `видимых панелей ${visiblePanels}`);
		check(
			`Tabs ${id}: вкладки и панели связаны по id`,
			tabs.every((item) => {
				const panel = document.getElementById(item.getAttribute('aria-controls') ?? '');
				return panel?.getAttribute('aria-labelledby') === item.id;
			}),
			`вкладок ${tabs.length}`
		);

		const selectedText = ratio(getComputedStyle(firstSelected).color);
		check(`Tabs ${id}: текст выбранной вкладки читаем (4.5:1)`, selectedText >= 4.5, `контраст ${selectedText.toFixed(2)}:1`);
		const plainText = ratio(getComputedStyle(plain).color);
		check(`Tabs ${id}: текст невыбранной вкладки читаем (4.5:1)`, plainText >= 4.5, `контраст ${plainText.toFixed(2)}:1`);

		const shadow = getComputedStyle(firstSelected).boxShadow;
		const indicator = COLOR_IN_SHADOW.exec(shadow)?.[0];
		const indicatorRatio = indicator === undefined ? 0 : ratio(indicator);
		check(`Tabs ${id}: индикатор выбранной вкладки отличим от фона (3:1)`, indicatorRatio >= 3, `контраст ${indicatorRatio.toFixed(2)}:1, box-shadow = ${shadow}`);
		check(`Tabs ${id}: у невыбранной вкладки индикатора нет`, getComputedStyle(plain).boxShadow === 'none', `box-shadow = ${getComputedStyle(plain).boxShadow}`);

		if (id === 'scroll') {
			check(
				`Tabs ${id}: содержимое шире полосы`,
				list.scrollWidth > list.clientWidth,
				`scrollWidth ${list.scrollWidth}px, clientWidth ${list.clientWidth}px`
			);
			check(`Tabs ${id}: прокрутка по горизонтали включена`, getComputedStyle(list).overflowX === 'auto', `overflow-x = ${getComputedStyle(list).overflowX}`);
			check(
				`Tabs ${id}: вкладки занимают всю высоту полосы`,
				tabs.every((item) => item.offsetHeight === list.clientHeight),
				`высота полосы ${list.clientHeight}px, вкладки ${[...new Set(tabs.map((item) => item.offsetHeight))].join(', ')}px`
			);
		}
	}
	return results;
}
```
