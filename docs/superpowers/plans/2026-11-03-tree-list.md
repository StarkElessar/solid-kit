# TreeList. План реализации

> **Для исполнителя:** REQUIRED SUB-SKILL: superpowers:executing-plans. Шаги с чекбоксами, задачи маленькие: каждая заканчивается зелёными тестами и своим коммитом. После каждой задачи: `pnpm vitest run src/components/grid src/components/tree-list` (код возврата отдельно), перед коммитом части: `pnpm check`. Любая правка вида или поведения идёт вместе с макетом (скил `solid-kit-design-sync`).

**Цель:** иерархическая таблица TreeList поверх движка грида с двумя вариантами подключения (плоский список с `parentKey`, ленивая догрузка детей), виртуализацией, закреплением колонок, четырьмя кнопками уровней и футером с пейджером.

**Спека:** `docs/superpowers/specs/2026-11-03-tree-list-design.md`.

**Архитектура:** движок грида `GridBase` общий; дерево подключается внутренним пропсом `tree` (`GridTreeSource`); все общие части (шапка, строки, ячейки, меню колонки, ресайз, футер, кнопки уровней, шеврон, полоса) остаются в `src/components/grid/` в одном экземпляре, в `src/components/tree-list/` лежит только то, чего в гриде нет: оболочка, построение и обход дерева, ленивая догрузка.

**Стек:** Solid 1.9, vitest 5 (jsdom), TS 5.9 strict (`noUncheckedIndexedAccess`, `exactOptionalPropertyTypes`), CSS Modules с нативной вложенностью.

## Глобальные ограничения

- Один класс, тип или функция на файл (kebab-case по имени), табы, одинарные кавычки, max-len 140, `simple-import-sort`, без non-null assertions, `--max-warnings 0`.
- CSS: нативная вложенность с явным `&`, токены `--s-kit-*`, на верхнем уровне только корневые селекторы блоков.
- Поведение грида не меняется: `grid.test.tsx` и остальные тесты грида проходят без правок.
- Не копировать файлы грида в `tree-list/`; не заводить `.module.css` в `tree-list/`.
- Публичный репозиторий: никаких ключей и данных компании (в демо вымышленные названия документов).
- Не пушить и не ставить теги без слова владельца.
- Коммит-сообщение заканчивается строками атрибуции из системной подсказки (`Co-Authored-By`, `Claude-Session`).

- [x] `build-tree-index.ts`:

```ts
import type { RowKey } from '../grid/row-key';

import type { TreeEntry } from './tree-entry';
import type { TreeIndex } from './tree-index';

/**
 * Индекс «родитель → дети» за один проход O(n). Родителя нет в списке: узел корневой. Цикл разрывается: первый его узел по порядку
 * `rows` становится корнем. Повтор ключа пропускается (как в `buildRowIndex`, учитывается первая строка).
 */
export function buildTreeIndex<T>(
	rows: readonly T[],
	getRowKey: (row: T) => RowKey,
	getParentKey: (row: T) => RowKey | null | undefined
): TreeIndex<T> {
	const entries: TreeEntry<T>[] = [];
	const byKey = new Map<RowKey, TreeEntry<T>>();
	for (const row of rows) {
		const key = getRowKey(row);
		if (!byKey.has(key)) {
			const entry = { row, key };
			byKey.set(key, entry);
			entries.push(entry);
		}
	}
	const roots: TreeEntry<T>[] = [];
	const children = new Map<RowKey, TreeEntry<T>[]>();
	const parentOf = new Map<RowKey, RowKey>();
	for (const entry of entries) {
		const parent = getParentKey(entry.row);
		if (parent === undefined || parent === null || parent === entry.key || !byKey.has(parent)) {
			roots.push(entry);
			continue;
		}
		parentOf.set(entry.key, parent);
		const siblings = children.get(parent);
		if (siblings === undefined) {
			children.set(parent, [entry]);
		}
		else {
			siblings.push(entry);
		}
	}
	// узлы цикла недостижимы из корней: первый из них по порядку отцепляется от родителя и становится корнем
	const reached = new Set<RowKey>();
	const mark = (start: TreeEntry<T>): void => {
		const stack = [start];
		for (let entry = stack.pop(); entry !== undefined; entry = stack.pop()) {
			if (!reached.has(entry.key)) {
				reached.add(entry.key);
				stack.push(...(children.get(entry.key) ?? []));
			}
		}
	};
	for (const root of roots) {
		mark(root);
	}
	for (const entry of entries) {
		if (reached.has(entry.key)) {
			continue;
		}
		const siblings = children.get(parentOf.get(entry.key) as RowKey);
		siblings?.splice(siblings.indexOf(entry), 1);
		roots.push(entry);
		mark(entry);
	}
	return { roots, children, byKey };
}
```

Примечание для исполнителя: `as RowKey` допустим здесь (у недостижимого узла родитель всегда есть); если линтер запрещает, заменить на проверку `const parent = parentOf.get(...)`, `if (parent !== undefined)`.

```ts
import type { FlatItems } from '../grid/flat-items';
import type { GridItem } from '../grid/grid-item';
import type { RowKey } from '../grid/row-key';

import type { TreeEntry } from './tree-entry';
import type { TreeIndex } from './tree-index';

interface Frame<T> {
	list: readonly TreeEntry<T>[];
	next: number;
	level: number;
}

/**
 * Дерево в плоский список видимых строк «родитель, затем его дети». Узел виден, если раскрыты все его предки; раскрыть можно только
 * раскрываемый узел (есть дети или `hasChildren`). Развёрнутый узел без детей, чей ключ в `loading`, получает заглушку. Обход
 * итеративный: глубокая цепочка не переполняет стек.
 */
export function flattenTree<T>(
	index: TreeIndex<T>,
	expanded: ReadonlySet<RowKey>,
	hasChildren: ((row: T) => boolean) | undefined,
	loading: ReadonlySet<RowKey>
): FlatItems<T> {
	const items: GridItem<T>[] = [];
	const rows: T[] = [];
	const stack: Frame<T>[] = [{ list: index.roots, next: 0, level: 0 }];
	for (let frame = stack[stack.length - 1]; frame !== undefined; frame = stack[stack.length - 1]) {
		const entry = frame.list[frame.next];
		if (entry === undefined) {
			stack.pop();
			continue;
		}
		const position = frame.next + 1;
		frame.next = position;
		const kids = index.children.get(entry.key);
		const expandable = kids !== undefined || hasChildren?.(entry.row) === true;
		const open = expandable && expanded.has(entry.key);
		items.push({
			kind: 'row',
			row: entry.row,
			rowIndex: rows.length,
			level: frame.level,
			node: { key: entry.key, expandable, expanded: open, setSize: frame.list.length, posInSet: position }
		});
		rows.push(entry.row);
		if (open) {
			if (kids !== undefined) {
				stack.push({ list: kids, next: 0, level: frame.level + 1 });
			}
			else if (loading.has(entry.key)) {
				items.push({ kind: 'loading', key: entry.key, level: frame.level + 1 });
			}
		}
	}
	return { items, rows };
}
```

// collapse
export function collapseTreeLevelKeys<T>(items: readonly GridItem<T>[], expanded: ReadonlySet<RowKey>): ReadonlySet<RowKey> {
	let deepest = -1;
	for (const item of items) {
		if (item.kind === 'row' && item.node?.expanded === true) {
			deepest = Math.max(deepest, item.level);
		}
	}
	if (deepest === -1) {
		return expanded;
	}
	const next = new Set(expanded);
	for (const item of items) {
		if (item.kind === 'row' && item.node?.expanded === true && item.level === deepest) {
			next.delete(item.node.key);
		}
	}
	return next;
}

// expand
export function expandTreeLevelKeys<T>(items: readonly GridItem<T>[], expanded: ReadonlySet<RowKey>): ReadonlySet<RowKey> {
	let shallowest = Number.POSITIVE_INFINITY;
	for (const item of items) {
		if (item.kind === 'row' && item.node?.expandable === true && !item.node.expanded) {
			shallowest = Math.min(shallowest, item.level);
		}
	}
	if (shallowest === Number.POSITIVE_INFINITY) {
		return expanded;
	}
	const next = new Set(expanded);
	for (const item of items) {
		if (item.kind === 'row' && item.node?.expandable === true && !item.node.expanded && item.level === shallowest) {
			next.add(item.node.key);
		}
	}
	return next;
}

// all
export function allTreeKeys<T>(index: TreeIndex<T>): ReadonlySet<RowKey> {
	return new Set(index.children.keys());
}

// loading
export function loadingNodeKeys<T>(/* сигнатура выше */): ReadonlySet<RowKey> {
	if (hasChildren === undefined) {
		return loading;
	}
	const next = new Set(loading);
	for (const key of expanded) {
		const entry = index.byKey.get(key);
		if (entry !== undefined && !index.children.has(key) && !requested.has(key) && hasChildren(entry.row)) {
			next.add(key);
		}
	}
	return next;
}

const NO_ROWS: readonly never[] = [];

export function TreeList<T>(props: TreeListProps<T>): JSX.Element {
	const [local, grid] = splitProps(props, [
		'rows', 'getParentKey', 'loadChildren', 'hasChildren', 'expandedKeys', 'defaultExpandedKeys', 'onExpandedKeysChange',
		'loadingKeys', 'levelLabels'
	]);
	const [expanded, setExpanded] = createControllableSignal<ReadonlySet<RowKey>>({
		name: 'TreeList',
		value: () => local.expandedKeys,
		defaultValue: () => local.defaultExpandedKeys ?? NO_KEYS,
		onChange: (next) => local.onExpandedKeysChange?.(next)
	});
	const index = createMemo(() => buildTreeIndex(local.rows ?? NO_ROWS, props.getRowKey, local.getParentKey ?? (() => undefined)));
	const flat = createMemo(() => flattenTree(index(), expanded(), local.hasChildren, local.loadingKeys ?? NO_KEYS));
	const change = (next: ReadonlySet<RowKey>): void => {
		if (next !== expanded()) {
			setExpanded(next);
		}
	};
	const source: GridTreeSource<T> = {
		flat,
		hasNodes: () => flat().items.length > 0,
		toggleNode: (key) => {
			change(toggleKey(expanded(), key));
		},
		loadNode: () => {},
		collapseLevel: () => {

		collapseAll: () => {
			change(expanded().size === 0 ? expanded() : NO_KEYS);
		},
		expandLevel: () => {
			change(expandTreeLevelKeys(flat().items, expanded()));
		},
		expandAll: () => {
			change(allTreeKeys(index()));
